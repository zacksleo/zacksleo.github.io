# 小艺智能体开发实战：创建单 Agent

## 前言

在前面的文章中，我们介绍了 A2A 协议的基础和消息规范。本文将详细介绍如何创建一个完整的单 Agent 应用，包括意图识别、业务逻辑处理和与大模型的集成。

## 单 Agent 架构设计

一个完整的单 Agent 应用包含以下核心模块：

```
┌─────────────────────────────────────────────────┐
│                   Client                        │

                      ▼

```

from enum import Enum

class IntentType(str, Enum):
    GENERATE = "generate"           # 生成内容
    CHAT = "chat"                   # 闲聊
    MODIFY = "modify"               # 修改内容
    REGENERATE = "regenerate"       # 重新生成
    CANCEL = "cancel"               # 取消任务
    CLEAR_CONTEXT = "clear_context" # 清空上下文
    UNKNOWN = "unknown"             # 未知意图
```

### 基于大模型的意图识别

```python
import json
import httpx
from typing import Tuple, Optional, Dict

INTENT_SYSTEM_PROMPT = """你是一个意图识别助手。请分析用户的输入，判断用户的意图。

可选的意图类型：
1. generate - 生成内容：用户想要创建新的内容
2. chat - 闲聊：问候、感谢、告别等社交对话
3. modify - 修改内容：修改已有内容的某些特征
4. regenerate - 重新生成：不喜欢上一个结果，想要重新生成
5. cancel - 取消：停止当前操作
6. clear_context - 清空上下文：重新开始
7. unknown - 未知：无法判断意图

请返回 JSON 格式：
- intent: 意图类型
- confidence: 置信度 (0-1)
- reason: 判断理由
- topic: 如果是 generate，提取主题；否则为空
"""

async def classify_intent_with_llm(
    user_text: str,
    context: Optional[Dict] = None,
    api_key: str = None,
    base_url: str = None
) -> Tuple[IntentType, float, str, str]:
    """使用大模型识别用户意图"""
    
    # 构建上下文信息
    has_previous = context.get('has_previous', False) if context else False
    context_info = f"之前已生成内容：{has_previous}"
    
    user_prompt = f"""用户输入：{user_text}
对话上下文：{context_info}

请分析用户意图并返回 JSON。"""
    
    async with httpx.AsyncClient(timeout=30.0) as client:
        response = await client.post(
            f"{base_url}/chat/completions",
            headers={
                "Content-Type": "application/json",
                "Authorization": f"Bearer {api_key}",
            },
            json={
                "model": "intent-model",
                "messages": [
                    {"role": "system", "content": INTENT_SYSTEM_PROMPT},
                    {"role": "user", "content": user_prompt}
                ],
                "temperature": 0.1,
                "max_tokens": 500
            }
        )
        
        if response.status_code == 200:
            data = response.json()
            result_text = data["choices"][0]["message"]["content"].strip()
            
            # 处理可能的 markdown 格式
            if result_text.startswith("```"):
                result_text = result_text.strip("```json").strip("```").strip()
            
            try:
                result = json.loads(result_text)
                intent_str = result.get("intent", "unknown")
                confidence = float(result.get("confidence", 0.5))
                reason = result.get("reason", "")
                topic = result.get("topic", "")
                
                # 字符串映射到枚举
                intent_map = {
                    "generate": IntentType.GENERATE,
                    "chat": IntentType.CHAT,
                    "modify": IntentType.MODIFY,
                    "regenerate": IntentType.REGENERATE,
                    "cancel": IntentType.CANCEL,
                    "clear_context": IntentType.CLEAR_CONTEXT,
                }
                intent_type = intent_map.get(intent_str, IntentType.UNKNOWN)
                
                return intent_type, confidence, f"LLM 判断：{reason}", topic
                
            except json.JSONDecodeError:
                return IntentType.UNKNOWN, 0.0, "LLM 返回格式错误", ""
        
        return IntentType.UNKNOWN, 0.0, f"LLM 调用失败：{response.status_code}", ""
```

## 业务服务层

### 服务层基础结构

```python
# services/assistant_service.py
import json
import httpx
from typing import List, Dict, AsyncGenerator

class AssistantService:
    """智能助手服务"""
    
    def __init__(self, api_key: str, base_url: str, model: str):
        self.api_key = api_key
        self.base_url = base_url
        self.model = model
    
    async def chat_with_llm_stream(
        self,
        user_text: str,
        context_messages: List[Dict] = None,
    ) -> AsyncGenerator[str, None]:
        """调用大模型进行对话（流式输出）"""
        
        system_prompt = """你是一个友好的智能助手。"""
        
        messages = [{"role": "system", "content": system_prompt}]
        
        if context_messages:
            messages.extend(context_messages[-6:])
        
        messages.append({"role": "user", "content": user_text})
        
        async with httpx.AsyncClient(timeout=60.0) as client:
            request_body = {
                "model": self.model,
                "messages": messages,
                "max_tokens": 512,
                "temperature": 0.7,
                "stream": True,
            }
            
            async with client.stream(
                "POST",
                f"{self.base_url}/chat/completions",
                headers={
                    "Content-Type": "application/json",
                    "Authorization": f"Bearer {self.api_key}",
                },
                json=request_body,
            ) as response:
                if response.status_code != 200:
                    error_text = await response.aread()
                    raise Exception(f"模型调用失败：{response.status_code}")
                
                async for line in response.aiter_lines():
                    line = line.strip()
                    if not line or not line.startswith("data:"):
                        continue
                    if line == "data: [DONE]":
                        break
                    
                    try:
                        data = json.loads(line[5:])
                        if "choices" in data and len(data["choices"]) > 0:
                            delta = data["choices"][0].get("delta", {})
                            content = delta.get("content", "")
                            if content:
                                yield content
                    except json.JSONDecodeError:
                        continue
```

### 内容生成服务

```python
    async def generate_content(
        self,
        user_prompt: str,
        context_messages: List[Dict] = None,
    ) -> AsyncGenerator[Dict, None]:
        """生成内容"""
        
        # 第一步：优化用户输入
        enhanced_prompt = await self.enhance_prompt(user_prompt, context_messages)
        yield {"type": "prompt", "content": enhanced_prompt}
        
        # 第二步：调用生成模型
        async for result in self.call_generation_model(enhanced_prompt):
            yield result
    
    async def enhance_prompt(
        self,
        user_prompt: str,
        context_messages: List[Dict] = None,
    ) -> str:
        """优化用户输入的提示词"""
        
        system_prompt = """你是一个专业的提示词优化专家。
用户会输入简单的描述，你需要将其优化为详细、具体的提示词。
要求：
1. 保持原意，添加更多细节
2. 包含风格、光线、色彩、构图等元素
3. 直接返回优化后的提示词，不要有其他说明
"""
        
        messages = [{"role": "system", "content": system_prompt}]
        if context_messages:
            messages.extend(context_messages[-10:])
        messages.append({
            "role": "user",
            "content": f"请优化以下提示词：{user_prompt}"
        })
        
        async with httpx.AsyncClient(timeout=60.0) as client:
            response = await client.post(
                f"{self.base_url}/chat/completions",
                headers={
                    "Content-Type": "application/json",
                    "Authorization": f"Bearer {self.api_key}",
                },
                json={
                    "model": self.model,
                    "messages": messages,
                    "max_tokens": 1024,
                    "temperature": 0.7,
                }
            )
            
            if response.status_code == 200:
                data = response.json()
                return data["choices"][0]["message"]["content"].strip()
        
        raise Exception("提示词优化失败")
    
    async def call_generation_model(
        self,
        prompt: str,
    ) -> AsyncGenerator[Dict, None]:
        """调用生成模型"""
        
        async with httpx.AsyncClient(timeout=120.0) as client:
            response = await client.post(
                f"{self.base_url}/generations",
                headers={
                    "Content-Type": "application/json",
                    "Authorization": f"Bearer {self.api_key}",
                },
                json={
                    "model": "generation-model",
                    "prompt": prompt,
                    "n": 1,
                }
            )
            
            if response.status_code == 200:
                data = response.json()
                if "data" in data and len(data["data"]) > 0:
                    yield {
                        "type": "result",
                        "url": data["data"][0].get("url"),
                        "prompt": prompt,
                    }
```

## 完整的 Agent 路由实现

```python
# routes/agent_routes.py
from fastapi import APIRouter, Request, Header
from fastapi.responses import JSONResponse, StreamingResponse
import uuid
import json
import asyncio
from typing import Optional, Dict, Any

from utils.intent_classifier import classify_intent_with_llm, IntentType, get_chat_response
from services.assistant_service import AssistantService
from core.config import settings

router = APIRouter(prefix="/agent", tags=["agent"])

# 共享状态
agent_sessions: Dict[str, Dict] = {}
conversation_contexts: Dict[str, list] = {}
task_states: Dict[str, Dict] = {}

# 服务实例
assistant_service = AssistantService(
    api_key=settings.VOLCANIC_API_KEY,
    base_url=settings.VOLCANIC_BASE_URL,
    model=settings.LANGUAGE_MODEL,
)

@router.post("/message")
async def handle_agent_message(
    request: Request,
    agent_session_id: Optional[str] = Header(None, alias="agent-session-id"),
):
    """统一的 Agent 消息处理接口"""
    request_id = str(uuid.uuid4())
    body = await request.json()
    jsonrpc_request = JsonRpcRequest(**body)
    
    method = jsonrpc_request.method
    
    if method == "message/stream":
        return await handle_message_stream(
            jsonrpc_request, agent_session_id, request_id
        )
    # ... 其他方法

async def handle_message_stream(
    request: JsonRpcRequest,
    agent_session_id: Optional[str],
    request_id: str,
) -> StreamingResponse:
    """处理 message/stream 方法"""
    
    params = request.params or {}
    task_id = params.get("id", str(uuid.uuid4()))
    session_id = params.get("sessionId", str(uuid.uuid4()))
    message = params.get("message", {})
    
    # 提取用户消息
    user_text = get_text_from_parts(message.get("parts", []))
    
    # 获取或创建对话上下文
    if session_id not in conversation_contexts:
        conversation_contexts[session_id] = []
    
    # 构建意图识别上下文
    intent_context = {
        "has_previous": any(
            msg.get("role") == "assistant"
            for msg in conversation_contexts[session_id][-5:]
        )
    }
    
    async def generate_sse():
        """SSE 流式生成器"""
        try:
            # 步骤 1: 发送 submitted 状态
            yield create_sse_event(
                task_id,
                "status-update",
                {"state": "submitted", "message": {"text": "正在理解您的需求..."}}
            )
            
            # 步骤 2: 意图识别
            intent_type, confidence, reason, topic = await classify_intent_with_llm(
                user_text, intent_context,
                settings.VOLCANIC_API_KEY,
                settings.VOLCANIC_BASE_URL
            )
            
            # 步骤 3: 根据意图类型处理
            if intent_type == IntentType.CHAT:
                async for chunk in assistant_service.chat_with_llm_stream(
                    user_text, conversation_contexts[session_id]
                ):
                    yield create_sse_event(
                        task_id,
                        "artifact-update",
                        {"artifactId": f"{task_id}-chat", "parts": [{"kind": "text", "text": chunk}]}
                    )
            
            elif intent_type == IntentType.GENERATE:
                async for result in assistant_service.generate_content(
                    user_text, conversation_contexts[session_id]
                ):
                    if result["type"] == "prompt":
                        yield create_sse_event(
                            task_id,
                            "artifact-update",
                            {"artifactId": f"{task_id}-prompt", 
                             "parts": [{"kind": "reasoningText", "reasoningText": result["content"]}]}
                        )
                    elif result["type"] == "result":
                        yield create_sse_event(
                            task_id,
                            "artifact-update",
                            {"artifactId": f"{task_id}-result",
                             "parts": [{"kind": "data", "data": result}]}
                        )
            
            # 步骤 4: 发送完成状态
            yield create_sse_event(
                task_id,
                "status-update",
                {"state": "completed", "message": {"text": "任务完成！"}}
            )
            
        except Exception as e:
            yield create_sse_event(
                task_id,
                "status-update",
                {"state": "failed", "message": {"text": f"错误：{str(e)}"}}
            )
    
    return StreamingResponse(
        generate_sse(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        }
    )

def create_sse_event(
    task_id: str,
    kind: str,
    data: Dict,
    final: bool = False,
) -> str:
    """创建 SSE 事件"""
    event = {
        "jsonrpc": "2.0",
        "id": task_id,
        "result": {
            "taskId": task_id,
            "kind": kind,
            "final": final,
            **data
        },
        "error": {"code": 0, "message": "success"}
    }
    return f"data: {json.dumps(event, ensure_ascii=False)}\n\n"
```
